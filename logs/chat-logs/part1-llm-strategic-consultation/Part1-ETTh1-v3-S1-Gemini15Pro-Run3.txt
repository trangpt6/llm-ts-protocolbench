import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.seasonal import seasonal_decompose

# Load the dataset
df = pd.read_csv('ETTh1.csv')

# Convert date to datetime
df['date'] = pd.to_datetime(df['date'])
df.set_index('date', inplace=True)

# Basic info
print(df.info())
print(df.head())

# The target is usually 'OT' in ETT datasets.
# Let's perform a seasonal decomposition to identify trend and seasonality.
# Since it's ETTh1 (hourly), we'll look for daily (24) or weekly (24*7) seasonality.
target = 'OT'
result = seasonal_decompose(df[target], model='additive', period=24)

# Plotting the components to visualize
fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(15, 12))
result.observed.plot(ax=ax1, title='Observed')
result.trend.plot(ax=ax2, title='Trend')
result.seasonal.plot(ax=ax3, title='Seasonality (Daily)')
result.resid.plot(ax=ax4, title='Residuals (Noise)')
plt.tight_layout()
plt.savefig('decomposition.png')

# Also check for weekly seasonality
