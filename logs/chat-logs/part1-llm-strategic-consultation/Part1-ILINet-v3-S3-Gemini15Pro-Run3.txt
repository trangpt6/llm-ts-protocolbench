import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.seasonal import seasonal_decompose

df = pd.read_csv('ILINet.csv')
df['DATE'] = pd.to_datetime(df['DATE'])
df.set_index('DATE', inplace=True)

# Resample or ensure frequency for decomposition
# The data is weekly.
target = df['% WEIGHTED ILI']

# Decomposition - assuming annual seasonality (52 weeks)
decomposition = seasonal_decompose(target, model='additive', period=52)

fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(12, 10), sharex=True)
decomposition.observed.plot(ax=ax1)
ax1.set_ylabel('Observed')
decomposition.trend.plot(ax=ax2)
ax2.set_ylabel('Trend')
decomposition.seasonal.plot(ax=ax3)
ax3.set_ylabel('Seasonal')
decomposition.resid.plot(ax=ax4)
ax4.set_ylabel('Residual')
plt.tight_layout()
plt.savefig('ili_decomposition.png')

# Statistical summary for noise
print(f"Mean: {target.mean()}")
print(f"Std: {target.std()}")
print(f"Residual Std: {decomposition.resid.std()}")
