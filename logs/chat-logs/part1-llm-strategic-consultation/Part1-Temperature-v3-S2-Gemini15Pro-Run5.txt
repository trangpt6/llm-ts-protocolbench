# Initial decomposition to see characteristics
# Assuming daily data based on the snippet (1/1/1981, 1/2/1981...)

try:
    result = seasonal_decompose(df[target_col], model='additive', period=365)

    fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(10, 10))
