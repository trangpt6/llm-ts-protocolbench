import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.seasonal import seasonal_decompose

# Load the dataset
df = pd.read_csv('Temperature.csv')

# Inspect the data
print(df.head())
print(df.info())

# Ensure Date is datetime and set as index
df['Date'] = pd.to_datetime(df['Date'])
df.set_index('Date', inplace=True)

# Identify the target column
target_col = 'Daily minimum temperatures'

# Basic analysis: Plotting and Decomposition
# Since it's daily data, common periods are 7 (weekly) or 365 (yearly).
# Let's check for seasonality using decomposition.
# Given daily temperatures, yearly seasonality is highly expected.
result = seasonal_decompose(df[target_col], model='additive', period=365)

# Plot components
fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(10, 8), sharex=True)
result.observed.plot(ax=ax1, title='Observed')
result.trend.plot(ax=ax2, title='Trend')
result.seasonal.plot(ax=ax3, title='Seasonal (Yearly)')
result.resid.plot(ax=ax4, title='Residual')
plt.tight_layout()
plt.savefig('decomposition.png')

# Print summary stats to help with characterization
print(df[target_col].describe())
