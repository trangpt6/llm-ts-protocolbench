import pandas as pd
import numpy as np
from statsmodels.tsa.seasonal import seasonal_decompose

df = pd.read_csv('ice_cream_heater.csv')
df['Month'] = pd.to_datetime(df['Month'])
df.set_index('Month', inplace=True)

print("Columns:", df.columns)
print("Data shape:", df.shape)

for col in df.columns:
    print(f"\n--- Analysis for {col} ---")
    # Decompose
    try:
        decomp = seasonal_decompose(df[col], model='additive', period=12)
        trend_strength = max(0, 1 - np.var(decomp.resid) / np.var(decomp.trend + decomp.resid))
        seasonal_strength = max(0, 1 - np.var(decomp.resid) / np.var(decomp.seasonal + decomp.resid))
        print(f"Trend strength: {trend_strength:.3f}")
        print(f"Seasonal strength: {seasonal_strength:.3f}")
        print(f"Residual variance: {np.var(decomp.resid):.3f}")
    except Exception as e:
        print("Decomp failed:", e)
