import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.seasonal import seasonal_decompose

# Load the dataset
df = pd.read_csv('Temperature.csv')

# Basic inspection
print(df.head())
print(df.info())

# Convert Date to datetime and set as index
# Need to handle potential parsing issues
df['Date'] = pd.to_datetime(df['Date'], errors='coerce')
df = df.dropna(subset=['Date'])
df.set_index('Date', inplace=True)

# Ensure the target column is numeric
target_col = 'Daily minimum temperatures'
df[target_col] = pd.to_numeric(df[target_col], errors='coerce')
df = df.dropna(subset=[target_col])

# Sorting by date
df.sort_index(inplace=True)

# Frequency inference
df = df.asfreq('D') # Daily data
df[target_col] = df[target_col].interpolate() # Handle missing days if any

# Decompose
# Using a period of 365 for annual seasonality
result = seasonal_decompose(df[target_col], model='additive', period=365)

# Plot for visual confirmation (internal use)
fig, (ax1,ax2,ax3,ax4) = plt.subplots(4,1, figsize=(15,10))
result.observed.plot(ax=ax1)
ax1.set_ylabel('Observed')
result.trend.plot(ax=ax2)
ax2.set_ylabel('Trend')
result.seasonal.plot(ax=ax3)
ax3.set_ylabel('Seasonal')
result.resid.plot(ax=ax4)
ax4.set_ylabel('Resid')
plt.tight_layout()
plt.savefig('decomposition.png')

print(df.describe())
