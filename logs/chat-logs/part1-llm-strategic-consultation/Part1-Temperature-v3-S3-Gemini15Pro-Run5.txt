# Seasonal decomposition assuming annual seasonality (approx 365 days)
# Using a 365-day period
result = seasonal_decompose(df['Daily minimum temperatures'], model='additive', period=365)

# Plotting decomposition components
fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(12, 10), sharex=True)
