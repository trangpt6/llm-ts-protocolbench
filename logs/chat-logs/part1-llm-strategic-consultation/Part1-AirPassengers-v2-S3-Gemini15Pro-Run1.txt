import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.seasonal import seasonal_decompose

# Load the dataset
df = pd.read_csv('AirPassengers.csv')

# Convert 'Month' to datetime and set as index
df['Month'] = pd.to_datetime(df['Month'])
df.set_index('Month', inplace=True)

# Basic inspection
print(df.head())
print(df.info())

# Decomposition to see trend, seasonality, and noise
# The data is known to have multiplicative seasonality (AirPassengers)
result = seasonal_decompose(df['#Passengers'], model='multiplicative', period=12)

# Plotting to confirm visually
fig, (ax1, ax2, ax3, ax4) = plt.subplots(4, 1, figsize=(10, 8), sharex=True)
result.observed.plot(ax=ax1, title='Observed')
result.trend.plot(ax=ax2, title='Trend')
result.seasonal.plot(ax=ax3, title='Seasonality')
result.resid.plot(ax=ax4, title='Residuals (Noise)')
plt.tight_layout()
plt.savefig('decomposition.png')

# Statistical properties
print("Trend detected:", "Yes" if result.trend.notnull().any() else "No")
